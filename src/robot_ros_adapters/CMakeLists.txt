cmake_minimum_required(VERSION 3.16)
project(robot_ros_adapters LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(robot_core REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(control_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(robot_ros_adapters SHARED src/adapters.cpp)
target_compile_features(robot_ros_adapters PUBLIC cxx_std_17)
target_compile_options(robot_ros_adapters PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(robot_ros_adapters PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
install(DIRECTORY include/ DESTINATION include)
ament_target_dependencies(robot_ros_adapters rclcpp rclcpp_action moveit_msgs control_msgs sensor_msgs std_msgs shape_msgs tf2_ros)
target_link_libraries(robot_ros_adapters robot_core::robot_core)
install(TARGETS robot_ros_adapters EXPORT export_robot_ros_adapters LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
ament_export_targets(export_robot_ros_adapters)
ament_export_dependencies(robot_core rclcpp rclcpp_action moveit_msgs control_msgs sensor_msgs std_msgs shape_msgs tf2_ros)
ament_package()
