cmake_minimum_required(VERSION 3.16)
project(robot_ros_adapters LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(robot_core REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(control_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(robot_interfaces REQUIRED)
find_package(trajectory_msgs REQUIRED)
add_library(robot_ros_adapters SHARED src/adapters.cpp)
target_compile_features(robot_ros_adapters PUBLIC cxx_std_17)
target_compile_options(robot_ros_adapters PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(robot_ros_adapters PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
install(DIRECTORY include/ DESTINATION include)
ament_target_dependencies(robot_ros_adapters rclcpp rclcpp_action moveit_msgs control_msgs sensor_msgs std_msgs shape_msgs tf2_ros geometry_msgs tf2_msgs trajectory_msgs robot_interfaces)
target_link_libraries(robot_ros_adapters robot_core::robot_core)
install(TARGETS robot_ros_adapters EXPORT export_robot_ros_adapters LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
ament_export_targets(export_robot_ros_adapters)
ament_export_dependencies(robot_core rclcpp rclcpp_action moveit_msgs control_msgs sensor_msgs std_msgs shape_msgs tf2_ros geometry_msgs tf2_msgs trajectory_msgs robot_interfaces)
if(BUILD_TESTING)
  enable_testing()
  add_executable(time_source_test test/time_source_test.cpp)
  target_include_directories(time_source_test PRIVATE include)
  target_link_libraries(time_source_test robot_core::robot_core)
  target_compile_options(time_source_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME simulation_time_contracts COMMAND time_source_test)
  add_executable(joint_position_test test/joint_position_test.cpp)
  target_include_directories(joint_position_test PRIVATE include)
  target_compile_features(joint_position_test PRIVATE cxx_std_17)
  target_compile_options(joint_position_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME panda_joint_position_roundoff COMMAND joint_position_test)
endif()
ament_package()
