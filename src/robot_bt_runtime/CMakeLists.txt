cmake_minimum_required(VERSION 3.16)
project(robot_bt_runtime LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(behaviortree_cpp REQUIRED)
find_package(robot_core REQUIRED)
find_package(robot_interfaces REQUIRED)
add_executable(robot_runtime src/runtime_node.cpp)
target_compile_features(robot_runtime PRIVATE cxx_std_17)
target_compile_options(robot_runtime PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(robot_runtime rclcpp rclcpp_action ament_index_cpp robot_interfaces behaviortree_cpp)
target_link_libraries(robot_runtime robot_core::robot_core)
install(TARGETS robot_runtime DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY trees DESTINATION share/${PROJECT_NAME})
ament_package()
