cmake_minimum_required(VERSION 3.16)
project(robot_core LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
add_library(robot_core INTERFACE)
target_compile_features(robot_core INTERFACE cxx_std_17)
target_include_directories(robot_core INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
install(DIRECTORY include/ DESTINATION include)
install(TARGETS robot_core EXPORT export_robot_core)
ament_export_targets(export_robot_core)
if(BUILD_TESTING)
  enable_testing()
  add_executable(core_test test/core_test.cpp)
  target_link_libraries(core_test robot_core)
  target_compile_options(core_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME core_contracts COMMAND core_test)
endif()
ament_package()
